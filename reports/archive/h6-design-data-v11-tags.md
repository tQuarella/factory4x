```ts
// Tag accuracy over every guarded node of every area (shipped data), the H3 "Tag accuracy" rule:
// Strong = single-type wave <= 80% of drones', Weak = more than drones', Even otherwise.
import { defaultData as data, guardianStatus } from '@factory4x/sim';
import { minSingle, probeWorld } from './waves.ts';

const out: string[] = [`Shipped data v${data.dataVersion}: area node | d / l / f | tags l / f`];
let right = 0;
let total = 0;
for (const a of data.map.areas) {
  if (a.tier < 1) continue;
  const seeds = a.layout === 'seed' ? [1, 2, 3] : [1];
  for (const seed of seeds) {
    const w = probeWorld(seed, data, a.id, false);
    w.nodes.forEach((n, i) => {
      if (n.kind !== 'pocket' && n.kind !== 'heart') return;
      const [d, l, f] = (['drone', 'lancer', 'flak'] as const).map((t) => minSingle(w, i, t, data));
      const tags = guardianStatus(w, i).tags;
      const judged = ([['lancer', l], ['flak', f]] as const).map(([t, v]) => {
        const tag = tags[t].tag;
        const want = d === null || v === null ? null : v * 100 <= d * 80 ? 'strong' : v > d ? 'weak' : 'even';
        total++;
        if (want === null || want === tag) right++;
        return want === null || want === tag ? tag : `${tag} (waves say ${want})`;
      });
      out.push(`| ${a.id}${a.layout === 'seed' ? `/${w.system.name}` : ''} | ${n.id} | ${[d, l, f].map((v) => v ?? '>80').join(' / ')} | ${judged.join(' / ')} |`);
    });
  }
}
out.push(`Tags that match the waves: ${right} of ${total}.`);
console.log(out.join('\n'));
```
